import re

def parse_duration(text: str) -> int:
    """Parse a duration string like '1h30m', '45s', '2h', or '1h5m10s' into total seconds.
    
    Accepts units h, m, and s in that specific order, each appearing at most once.
    Raises ValueError on empty input or any format that does not strictly match the rules.
    """
    if not text:
        raise ValueError("Input cannot be empty")
    
    match = re.match(r'^(\d+)h?(\d+m)?(\d+s)?$', text)
    if not match:
        raise ValueError(f"Invalid duration format: {text!r}")
        
    h_str, m_str, s_str = match.groups()
    
    hours = int(h_str[:-1]) if h_str else 0
    minutes = int(m_str[:-1]) if m_str else 0
    seconds = int(s_str[:-1]) if s_str else 0
    
    return hours * 3600 + minutes * 60 + seconds
