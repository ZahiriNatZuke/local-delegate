import re

def parse_duration(text: str) -> int:
    """
    Parse a duration string like '1h30m', '45s', '2h', or '1h5m10s' into seconds.
    
    Accepts units h, m, and s in that order, each at most once.
    Raises ValueError on empty input or invalid format.
    """
    if not text:
        raise ValueError("Input cannot be empty")
    
    match = re.fullmatch(r'(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?', text)
    if not match:
        raise ValueError(f"Invalid duration format: {text!r}")
    
    hours, minutes, seconds = match.groups()
    if hours is None and minutes is None and seconds is None:
        raise ValueError("At least one unit (h, m, or s) is required")
    
    total_seconds = 0
    if hours is not None:
        total_seconds += int(hours) * 3600
    if minutes is not None:
        total_seconds += int(minutes) * 60
    if seconds is not None:
        total_seconds += int(seconds)
    
    return total_seconds
