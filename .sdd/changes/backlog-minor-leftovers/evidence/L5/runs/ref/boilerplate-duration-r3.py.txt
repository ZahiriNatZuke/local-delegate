import re

def parse_duration(text: str) -> int:
    """
    Parse a duration string like '1h30m', '45s', '2h', or '1h5m10s' into total seconds.
    
    Accepts units 'h' (hours), 'm' (minutes), and 's' (seconds) in that order,
    each appearing at most once. Raises ValueError on empty input or invalid format.
    
    Args:
        text: A string representing a duration.
        
    Returns:
        The total duration in seconds as an integer.
        
    Raises:
        ValueError: If the input is empty or does not match the expected format.
    """
    if not text:
        raise ValueError("Input string is empty")
    
    # Pattern: optional hours, optional minutes, optional seconds, each with a number
    # Units must appear in order: h, then m, then s
    # Each unit can appear at most once
    pattern = r'^(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?$'
    match = re.match(pattern, text)
    
    if not match:
        raise ValueError(f"Invalid duration format: '{text}'")
    
    hours_str, minutes_str, seconds_str = match.groups()
    
    # Check if any unit appears more than once by verifying no duplicate units in original string
    # The regex already enforces order and single occurrence per unit type due to structure
    # But we need to ensure that if a unit is present, it's correctly parsed
    
    hours = int(hours_str) if hours_str else 0
    minutes = int(minutes_str) if minutes_str else 0
    seconds = int(seconds_str) if seconds_str else 0
    
    # Validate that at least one unit was specified
    if hours_str is None and minutes_str is None and seconds_str is None:
        raise ValueError(f"Invalid duration format: '{text}'")
    
    total_seconds = hours * 3600 + minutes * 60 + seconds
    
    return total_seconds
