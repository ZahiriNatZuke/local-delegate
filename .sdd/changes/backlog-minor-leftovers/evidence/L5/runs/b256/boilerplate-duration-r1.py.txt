import re

def parse_duration(text: str) -> int:
    """Parse a duration string like '1h30m', '45s', '2h', or '1h5m10s' into seconds.
    
    Accepts h, m, and s units in that order, each at most once.
    Raises ValueError on empty input or invalid format.
    """
    if not text:
        raise ValueError("Input cannot be empty")
    
    pattern = r'^(?:(\d+)h)?(?:(\d+)m)?(?:(\d+)s)?$'
    match = re.match(pattern, text)
    
    if not match:
        raise ValueError(f"Invalid duration format: {text}")
        
    hours, minutes, seconds = match.groups()
    
    return (
        (int(hours) * 3600 if hours else 0) +
        (int(minutes) * 60 if minutes else 0) +
        (int(seconds) if seconds else 0)
    )
