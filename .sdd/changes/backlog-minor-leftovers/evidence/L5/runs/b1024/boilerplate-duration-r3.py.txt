import re

def parse_duration(text: str) -> int:
    """
    Parse a duration string like '1h30m', '45s', '2h', or '1h5m10s' into seconds.
    
    Accepts units h, m, and s in that specific order, each at most once.
    Raises ValueError on empty input or invalid format.
    """
    if not text:
        raise ValueError("Input cannot be empty")
    
    match = re.fullmatch(r'(\d+)h?(\d+m?)?(\d+s)?', text)
    if not match:
        raise ValueError(f"Invalid duration format: {text!r}")
    
    h_str, m_part, s_part = match.groups()
    
    h = int(h_str) if h_str else 0
    m = int(m_part[:-1]) if m_part and m_part[-1] == 'm' else (int(m_part) if m_part else 0)
    s = int(s_part[:-1]) if s_part and s_part[-1] == 's' else (int(s_part) if s_part else 0)
    
    return h * 3600 + m * 60 + s
```
